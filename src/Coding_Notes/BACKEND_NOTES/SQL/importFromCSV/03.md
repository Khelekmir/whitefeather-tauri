Great — here's a **generic CSV importer** that will let you reuse the same logic for **any table**, as long as you provide:

1. The table name
2. A list of column names (in order)
3. The CSV file path
4. The data types for each column (to help parse correctly)

---

### ✅ Generic CSV Importer

```java
package practice.feproj.backend.utility;

import java.io.BufferedReader;
import java.io.IOException;
import java.nio.file.*;
import java.sql.*;
import java.util.*;

public class GenericCsvImporter {

    public static void importCsvToTable(String tableName, List<String> columns, List<String> types, String csvPathStr) {
        Path csvPath = Paths.get(csvPathStr);
        String placeholders = String.join(", ", Collections.nCopies(columns.size(), "?"));
        String columnNames = String.join(", ", columns);

        String insertSQL = String.format(
            "INSERT INTO %s (%s) VALUES (%s)",
            tableName, columnNames, placeholders
        );

        try (
            Connection conn = DatabaseConnector.createConnection();
            BufferedReader reader = Files.newBufferedReader(csvPath);
            PreparedStatement ps = conn.prepareStatement(insertSQL)
        ) {
            conn.setAutoCommit(false);

            String line;
            int lineNumber = 0;

            while ((line = reader.readLine()) != null) {
                lineNumber++;
                if (line.trim().isEmpty()) continue;

                List<String> fields = parseCsvLine(line);
                if (fields.size() != columns.size()) {
                    System.err.printf("Skipping line %d: expected %d fields but got %d\n", lineNumber, columns.size(), fields.size());
                    continue;
                }

                for (int i = 0; i < fields.size(); i++) {
                    String value = fields.get(i).trim();

                    if (value.isEmpty()) {
                        ps.setNull(i + 1, Types.NULL);
                        continue;
                    }

                    switch (types.get(i).toLowerCase()) {
                        case "string" -> ps.setString(i + 1, value);
                        case "int", "integer" -> ps.setInt(i + 1, Integer.parseInt(value));
                        case "double", "float" -> ps.setDouble(i + 1, Double.parseDouble(value));
                        case "boolean" -> ps.setBoolean(i + 1, Boolean.parseBoolean(value));
                        case "json", "jsonb" -> ps.setObject(i + 1, value, Types.OTHER);
                        default -> throw new IllegalArgumentException("Unsupported type: " + types.get(i));
                    }
                }

                ps.addBatch();
            }

            ps.executeBatch();
            conn.commit();
            System.out.printf("✅ Imported CSV into %s successfully.\n", tableName);

        } catch (SQLException | IOException | IllegalArgumentException e) {
            System.err.println("❌ Error during CSV import:");
            e.printStackTrace();
        }
    }

    private static List<String> parseCsvLine(String line) {
        List<String> result = new ArrayList<>();
        StringBuilder current = new StringBuilder();
        boolean inQuotes = false;

        for (char c : line.toCharArray()) {
            if (c == ',' && !inQuotes) {
                result.add(current.toString());
                current.setLength(0);
            } else if (c == '"') {
                inQuotes = !inQuotes;
            } else {
                current.append(c);
            }
        }

        result.add(current.toString());
        return result;
    }
}
```

### ✅ Example Usage (for your `armor` table)

```java
public static void main(String[] args) {
    String csvPath = "backend/src/main/resources/data/armor.csv";

    List<String> columns = List.of(
        "item_name", "armor_type", "equipment_slot", "armor_description",
        "dodge_bonus", "parry_bonus", "block_bonus", "armor_bonus",
        "resist_bonus", "allure_bonus", "charisma_bonus",
        "armor_durability", "armor_degradation_rate",
        "flags", "armor_weight"
    );

    List<String> types = List.of(
        "string", "string", "string", "string",
        "double", "double", "double", "double",
        "double", "double", "double",
        "double", "double",
        "jsonb", "double"
    );

    GenericCsvImporter.importCsvToTable("public.armor", columns, types, csvPath);
}
```

### 🧠 Notes

* You must maintain the same order of columns and types in the lists and in the CSV file.
* JSON fields in the CSV should be *valid JSON*, e.g.:

```csv
"{"prf":"Amberyl","indestructible":true,"unequipped":true}"
```

* To make this more advanced later, you could:

  * Add support for headers in the CSV
  * Read column definitions from a config file
  * Auto-detect data types (less safe, more flexible)
